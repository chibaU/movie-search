// movie-search/src/App.jsx
import React, { useState } from "react";
import MovieCard from "./components/MovieCard";
import MovieCards from "./components/MovieCards";
import Search from "./components/Search";

const App = () => {
  const [title, setTitle] = useState("");

  return (
    <>
      <div className="">
        <h1>Movie search app</h1>
        <Search setSearchTitle={setTitle} />
        <MovieCards title={title} />
      </div>
    </>
  );
};

export default App;


// movie-search/src/components/Search.jsx
import React, { useState } from "react";

const Search = ({ setSearchTitle }) => {
  const [title, setTitle] = useState("");
  const [message, setMessage] = useState("");
  const onSubmitHandle = (e) => {
    setMessage("");
    e.preventDefault();
    if (title.trim().length < 3) {
      setMessage("Please enter at least 3 characters.");
      return;
    }

    setSearchTitle(title);
  };

  return (
    <>
      <section className="">
        <form onSubmit={onSubmitHandle} className="flex flex-row gap-2 p-4">
          <label className="px-2 py-2" htmlFor="title">
            Title
          </label>
          <input
            className="px-4 py-2 rounded-xl"
            type="text"
            placeholder="Title ..."
            name="Title"
            id="title"
            value={title}
            onChange={(e) => setTitle(e.target.value)}
          />

          <input
            className="px-4 py-2 bg-green-400 rounded-xl"
            type="submit"
            value="Search"
          />
        </form>
        <p className="text-red-500">{message}</p>
      </section>
    </>
  );
};

export default Search;


// movie-search/src/components/MovieCards.jsx
import React, { useEffect, useState } from "react";
import MovieCard from "./MovieCard";

const MovieCards = ({ title }) => {
  const [movies, setMovies] = useState([]);
  const [loading, setLoading] = useState(false);
  const [message, setMessage] = useState("");

  const search = title || "movie";
  const url = `https://www.omdbapi.com/?s=${search}&y=2026&page=1&apikey=4706733c`;

  useEffect(() => {
    // eslint-disable-next-line react-hooks/set-state-in-effect
    setLoading(true);
    fetch(url)
      .then((res) => res.json())
      .then((data) => {
        if (data.Response === "True") {
          setMovies(data.Search);
        } else {
          // هنا نعالج كل الحالات السيئة
          if (data.Error === "Too many results.") {
            setMessage("⚠️ النتائج كثيرة جدًا، حاول كتابة اسم أدق");
          } else {
            setMessage("❌ لا توجد نتائج لهذا البحث");
          }
        }

        setMovies(data.Search);
        console.log(data);
        setLoading(false);
      })
      .catch((err) => console.error(err));
  }, [url]);

  if (loading) {
    return <p className="text-center mt-5">⏳ جاري البحث...</p>;
  }

  if (message) {
    return <p className="text-center mt-5 text-yellow-500">{message}</p>;
  }

  return (
    <div className="grid grid-cols-3 gap-5">
      {movies.map((movie) => {
        return <MovieCard key={movie.imdbID} movie={movie} />;
      })}
    </div>
  );
};

export default MovieCards;

// movie-search/src/components/MovieCard.jsx
import movieFallback from "../assets/movie-fallback.svg";

const MovieCard = ({ movie }) => {
  const card_hover = "transition-transform duration-200 hover:scale-105";
  const poster =
    movie.Poster && movie.Poster !== "N/A" ? movie.Poster : movieFallback;

  return (
    <>
      <div
        className={`bg-white rounded shadow-lg p-2 w-52 cursor-pointer ${card_hover}`}
      >
        <div className="w-full aspect-2/3 overflow-hidden rounded">
          <img
            src={poster}
            alt={movie.Title}
            onError={(e) => {
              e.currentTarget.onerror = null;
              e.currentTarget.src = movieFallback;
            }}
            className="w-full h-full object-cover"
          />
        </div>
        <div className="mt-2">
          <h3 className="font-bold text-sm md:text-base">{movie.Title}</h3>
          <p className="text-gray-500 text-xs md:text-sm">{movie.Year}</p>
        </div>
      </div>
    </>
  );
};

export default MovieCard;
